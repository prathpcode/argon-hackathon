const AudioEngine = {
  ctx: null,
  muted: false,
  musicTimer: null,
  currentTrack: null,
  init() {
    if (!this.ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      if (AC) this.ctx = new AC();
    }
    if (this.ctx && this.ctx.state === "suspended") this.ctx.resume();
  },
  playTone(freq, dur = 0.25, type = "sine", gainVal = 0.12, delay = 0) {
    if (this.muted) return;
    this.init();
    if (!this.ctx) return;
    const now = this.ctx.currentTime + delay;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, now);
    gain.gain.setValueAtTime(0.001, now);
    gain.gain.linearRampToValueAtTime(gainVal, now + 0.02);
    gain.gain.exponentialRampToValueAtTime(0.0001, now + dur);
    osc.connect(gain);
    gain.connect(this.ctx.destination);
    osc.start(now);
    osc.stop(now + dur + 0.02);
  },
  playChime(kind = "quoin") {
    if (kind === "quoin") {
      this.playTone(587.33, 0.18, "triangle", 0.12, 0);
      this.playTone(880.0, 0.28, "triangle", 0.12, 0.08);
    } else if (kind === "hi") {
      this.playTone(523.25, 0.15, "sine", 0.14, 0);
      this.playTone(659.25, 0.15, "sine", 0.14, 0.1);
      this.playTone(1046.5, 0.35, "triangle", 0.16, 0.2);
    } else if (kind === "giant") {
      [220, 277.18, 329.63, 440, 659.25].forEach((f, i) => {
        this.playTone(f, 0.8, "triangle", 0.1, i * 0.09);
      });
    } else if (kind === "mine") {
      this.playTone(180, 0.09, "square", 0.08, 0);
      this.playTone(320, 0.12, "triangle", 0.08, 0.05);
    } else if (kind === "GNG") {
      // Melancholic lullaby melody ("Lay all things..." from Video A 50:39)
      const notes = [392, 440, 493.88, 587.33, 523.25, 493.88, 440, 392, 329.63, 349.23, 392];
      notes.forEach((f, i) => this.playTone(f, 0.45, "sine", 0.13, i * 0.42));
    } else if (kind === "DG2") {
      const notes = [523.25, 659.25, 783.99, 659.25, 880, 783.99, 659.25, 523.25];
      notes.forEach((f, i) => this.playTone(f, 0.22, "triangle", 0.12, i * 0.2));
    } else if (kind === "thriller") {
      // Funky bassline for Pumpkin Ale Thriller spell (Video A 29:43 / 32:20)
      const bass = [146.83, 164.81, 174.61, 196.0, 174.61, 164.81, 146.83, 130.81];
      bass.forEach((f, i) => this.playTone(f, 0.22, "sawtooth", 0.09, i * 0.24));
    }
  },
  startAmbient(style) {
    if (this.musicTimer) clearInterval(this.musicTimer);
    this.currentTrack = style;
    const scales = {
      lost_woods: [220, 261.63, 293.66, 329.63, 392.0, 440.0],
      vorn_road: [261.63, 329.63, 392.0, 493.88, 523.25, 659.25],
      savanna: [293.66, 329.63, 392.0, 440.0, 493.88, 587.33],
      cavern: [196.0, 233.08, 261.63, 293.66, 349.23],
      office: [261.63, 293.66, 349.23, 392.0]
    };
    const notes = scales[style] || scales.savanna;
    let step = 0;
    this.musicTimer = setInterval(() => {
      if (this.muted || !this.ctx || this.ctx.state !== "running") return;
      const freq = notes[step % notes.length];
      this.playTone(freq, 0.65, "sine", 0.035, 0);
      step = (step + 1 + Math.floor(seededRandom() * 2)) % notes.length;
    }, 950);
  }
};

// --- WebSocket Multiplayer Connection ---
function connectWebSocket() {
  try {
    const proto = location.protocol === "https:" ? "wss:" : "ws:";
    const ws = new WebSocket(`${proto}//${location.host}`);
    State.ws = ws;
    ws.onopen = () => {
      State.connected = true;
      ws.send(JSON.stringify({
        type: "join",
        id: State.playerId,
        name: State.player.name,
        street: State.streetId,
        x: State.player.x,
        y: State.player.y,
        dir: State.player.dir,
        scale: State.player.scale,
        appearance: State.player.appearance
      }));
      logTelemetry("player_join", { id: State.playerId, street: State.streetId, online: true });
    };
    ws.onmessage = (ev) => {
      try {
        const msg = JSON.parse(ev.data);
        handleServerMessage(msg);
      } catch (e) {}
    };
    ws.onclose = () => {
      State.connected = false;
    };
  } catch (e) {
    State.connected = false;
  }
}

function sendNet(payload) {
  if (State.connected && State.ws && State.ws.readyState === WebSocket.OPEN) {
    State.ws.send(JSON.stringify(payload));
  }
}

function handleServerMessage(msg) {
  if (msg.type === "street_players") {
    // Keep bots if no remote players, plus sync real remote players
    Object.keys(State.otherPlayers).forEach(k => {
      if (!State.otherPlayers[k].isBot) delete State.otherPlayers[k];
    });
    (msg.players || []).forEach(p => {
      if (p.id !== State.playerId) {
        State.otherPlayers[p.id] = p;
      }
    });
    updateOnlineListUI();
  } else if (msg.type === "player_moved") {
    if (msg.id !== State.playerId && msg.street === State.streetId) {
      State.otherPlayers[msg.id] = { ...(State.otherPlayers[msg.id] || {}), ...msg };
    }
  } else if (msg.type === "player_left") {
    delete State.otherPlayers[msg.id];
    updateOnlineListUI();
  } else if (msg.type === "chat") {
    addChatMessage(msg.sender, msg.text, msg.kind || "normal");
    if (msg.senderId) {
      State.chatBubbles[msg.senderId] = { text: msg.text, timer: 260 };
    }
  } else if (msg.type === "emote") {
    handleRemoteEmote(msg);
  } else if (msg.type === "street_style") {
    if (STREETS[msg.street]) {
      STREETS[msg.street].style = msg.style;
      if (State.streetId === msg.street) {
        AudioEngine.startAmbient(msg.style);
      }
    }
  } else if (msg.type === "tower_progress") {
    const res = STREETS.beau_home.resources.find(r => r.id === "tower_1");
    if (res) res.progress = msg.progress;
  } else if (msg.type === "co_mine_bonus") {
    addFloatingText(State.player.x, State.player.y - 65, "+1 Bonus Beryl (Co-op Mine!)", "#2ecc71");
    giveItem("beryl_chunk", 1);
  } else if (msg.type === "giant_visage") {
    const g = GIANTS_DATA.find(x => x.name === msg.giant) || GIANTS_DATA[0];
    showGiantOverlay(g);
  }
}

// --- Chat & Floating Feedback Helpers ---
function addChatMessage(sender, text, kind = "normal") {
  const box = document.getElementById("chat-messages");
  const div = document.createElement("div");
  div.className = "chat-msg " + kind;
  if (sender) {
    div.innerHTML = `<strong>${escapeHtml(sender)}:</strong> ${escapeHtml(text)}`;
  } else {
    div.textContent = text;
  }
  box.appendChild(div);
  box.scrollTop = box.scrollHeight;
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
}

function addFloatingText(x, y, text, color = "#ffffff") {
  State.floatingTexts.push({ x, y, text, color, life: 85, maxLife: 85 });
}

function spawnParticles(x, y, count = 20, kind = "sparkle") {
  for (let i = 0; i < count; i++) {
    const ang = seededRandom() * Math.PI * 2;
    const spd = 1 + seededRandom() * 4;
    State.particles.push({
      x, y,
      vx: Math.cos(ang) * spd,
      vy: Math.sin(ang) * spd - 1.5,
      kind,
      life: 55 + Math.floor(seededRandom() * 35),
      color: kind === "rainbow"
        ? ["#ff4b4b", "#ffa64b", "#ffe84b", "#59d959", "#4bb5ff", "#a66bff"][i % 6]
        : "#ffffff"
    });
  }
}

// --- Inventory & Stat Helpers ---
function changeMeters({ energy = 0, mood = 0, img = 0, currants = 0 }) {
  const p = State.player;
  if (energy !== 0) p.energy = Math.max(0, Math.min(p.maxEnergy, p.energy + energy));
  if (mood !== 0) p.mood = Math.max(0, Math.min(p.maxMood, p.mood + mood));
  if (img !== 0) {
    // GUESS: High mood (>80%) grants a 1.2x iMG bonus
    const mult = (p.mood / p.maxMood) >= 0.8 ? 1.2 : 1.0;
    p.img += Math.round(img * mult);
  }
  if (currants !== 0) p.currants = Math.max(0, p.currants + currants);
  updateHUD();
  logTelemetry("meter_change", { energy: p.energy, mood: p.mood, img: p.img, currants: p.currants });
}

function giveItem(itemId, amount = 1) {
  const def = ITEM_DEFS[itemId];
  if (!def) return false;
  // Try stacking first
  if (def.stack > 1) {
    const existing = State.inventory.find(s => s && s.id === itemId && s.count < def.stack);
    if (existing) {
      existing.count = Math.min(def.stack, existing.count + amount);
      renderInventory();
      return true;
    }
  }
  // Check if bag is full (16 slots max in main bar, Video B 03:51)
  if (State.inventory.length >= 16) {
    addFloatingText(State.player.x, State.player.y - 60, "Your bag is full!", "#e74c3c");
    addChatMessage(null, "Your bag is full! Drop or consume an item first.", "sys");
    return false;
  }
  State.inventory.push({ id: itemId, count: amount });
  renderInventory();
  return true;
}

function removeItem(itemId, amount = 1, fromBag = false) {
  const list = fromBag ? State.bagInventory : State.inventory;
  const idx = list.findIndex(s => s && s.id === itemId);
  if (idx === -1) return false;
  list[idx].count -= amount;
  if (list[idx].count <= 0) list.splice(idx, 1);
  renderInventory();
  return true;
}

function drawItemIconToCanvas(canvas, itemId) {
  const ctx = canvas.getContext("2d");
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  const def = ITEM_DEFS[itemId];
  if (!def) return;
  ctx.fillStyle = def.color;
  ctx.beginPath();
  ctx.arc(16, 16, 13, 0, Math.PI * 2);
  ctx.fill();
  ctx.font = "16px sans-serif";
  ctx.textAlign = "center";
  ctx.textBaseline = "middle";
  ctx.fillText(def.iconText || def.icon, 16, 17);
}

function renderInventory() {
  const row = document.getElementById("inv-slots-row");
  row.innerHTML = "";
  const filter = State.invFilter.trim().toLowerCase();

  for (let i = 0; i < 16; i++) {
    const slotData = State.inventory[i];
    const slotEl = document.createElement("div");
    slotEl.className = "inv-slot";
    if (slotData) {
      const def = ITEM_DEFS[slotData.id];
      if (filter && !def.name.toLowerCase().includes(filter)) {
        slotEl.classList.add("filtered-out");
      }
      const c = document.createElement("canvas");
      c.width = 32; c.height = 32;
      drawItemIconToCanvas(c, slotData.id);
      slotEl.appendChild(c);

      if (slotData.count > 1) {
        const cnt = document.createElement("span");
        cnt.className = "slot-count";
        cnt.textContent = slotData.count;
        slotEl.appendChild(cnt);
      }
      const tip = document.createElement("div");
      tip.className = "slot-name";
      tip.textContent = def.name;
      slotEl.appendChild(tip);

      slotEl.onclick = (e) => {
        e.stopPropagation();
        openItemContextMenu(slotData, false, e.clientX, e.clientY - 120);
      };
    }
    row.appendChild(slotEl);
  }

  // Bag sub-bar
  const bagBar = document.getElementById("bag-sub-bar");
  const bagRow = document.getElementById("bag-slots-row");
  bagBar.classList.toggle("open", State.bagOpen);
  bagRow.innerHTML = "";
  if (State.bagOpen) {
    for (let i = 0; i < 8; i++) {
      const slotData = State.bagInventory[i];
      const slotEl = document.createElement("div");
      slotEl.className = "inv-slot";
      slotEl.style.width = "46px";
      slotEl.style.height = "44px";
      if (slotData) {
        const def = ITEM_DEFS[slotData.id];
        const c = document.createElement("canvas");
        c.width = 32; c.height = 32;
        drawItemIconToCanvas(c, slotData.id);
        slotEl.appendChild(c);
        if (slotData.count > 1) {
          const cnt = document.createElement("span");
          cnt.className = "slot-count";
          cnt.textContent = slotData.count;

        slotEl.onclick = (e) => {
          e.stopPropagation();
          openItemContextMenu(slotData, true, e.clientX, e.clientY - 110);
        };
      }
      bagRow.appendChild(slotEl);
    }
  }
}

function advanceQuest(questId, delta = 1) {
  const q = QUESTS_LIST.find(x => x.id === questId);
  if (q && q.progress < q.goal) {

    if (q.progress === q.goal) {
      changeMeters({ currants: 100, img: 150, energy: 25, mood: 15 });
      addChatMessage(null, `🎉 Quest Completed: "${q.title}"! Reward: ${q.reward}`, "reward");
    }
  }
}

function formatDuration(sec) {
  const d = Math.floor(sec / 86400);
  const h = Math.floor((sec % 86400) / 3600);
  const m = Math.floor((sec % 3600) / 60);
  const s = sec % 60;
  if (d > 0) return `${d}d ${h}h ${m}m`;
  if (h > 0) return `${h}h ${m}m ${s}s`;
  return `${m}m ${s}s`;
}

function drawGiantTotem(canvas, giant) {
  const ctx = canvas.getContext("2d");
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.save();
  ctx.translate(90, 110);
  // Ornate symmetrical totem mask inspired by Cosma/Zille/Humbaba/Alph in Video A
  for (let i = 0; i < 12; i++) {
    ctx.save();
    ctx.rotate((i / 12) * Math.PI * 2);
    ctx.fillStyle = i % 2 === 0 ? giant.color : giant.accent;
    ctx.beginPath();
    ctx.ellipse(0, -68, 10, 26, 0, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
  }
  ctx.fillStyle = "#fff8e7";
  ctx.beginPath();
  ctx.ellipse(0, 0, 48, 62, 0, 0, Math.PI * 2);
  ctx.fill();
  ctx.lineWidth = 4;
  ctx.strokeStyle = giant.accent;
  ctx.stroke();

  // Eyes
  [-18, 18].forEach(ex => {
    ctx.fillStyle = "#2c3e50";
    ctx.beginPath();
    ctx.arc(ex, -10, 10, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = "#fff";
    ctx.beginPath();
    ctx.arc(ex - 3, -13, 3, 0, Math.PI * 2);
    ctx.fill();
  });
  // Ornate mouth/beard
  ctx.fillStyle = giant.accent;
  ctx.beginPath();
  ctx.arc(0, 22, 16, 0, Math.PI);
  ctx.fill();
  ctx.restore();
}

function changeStreet(targetStreetId, spawnX = 300) {
  const st = STREETS[targetStreetId];
  if (!st) return;
  State.waitingRoom.active = false;
  document.getElementById("waiting-banner").style.display = "none";

  triggerLoadingTransition("Entering", st.name, () => {
    State.streetId = targetStreetId;
    State.player.x = spawnX;
    State.player.y = st.groundY;
    State.player.vx = 0;
    State.player.vy = 0;

    // Move any following bots to the new street with us (Video A 26:48)
    State.player.followers.forEach((botId, idx) => {
      if (State.otherPlayers[botId]) {
        State.otherPlayers[botId].street = targetStreetId;
        State.otherPlayers[botId].x = spawnX - (idx + 1) * 45;
        State.otherPlayers[botId].y = st.groundY;
      }
    });

    if (!visitedStreets.has(targetStreetId)) {
      visitedStreets.add(targetStreetId);
      changeMeters({ img: 2 });
      addChatMessage(null, `This is your first visit to ${st.name}. You received 2 xp!`, "reward");
    } else {
      addChatMessage(null, `Entered ${st.name}.`, "sys");
    }

    AudioEngine.startAmbient(st.style);
    updateHUD();
    updateOnlineListUI();
    sendNet({
      type: "change_street",
      id: State.playerId,
      street: targetStreetId,
      x: State.player.x,
      y: State.player.y
    });
    logTelemetry("street_change", { street: targetStreetId, name: st.name });
  });
}

// --- Fixed-Timestep Simulation (60 Hz) ---
let simTick = 0;

function updateSimulation() {
  simTick++;
  const st = STREETS[State.streetId];
  const p = State.player;

  // Skill countdown every 60 ticks (1 sec)
  if (simTick % 60 === 0) {
    const sk = SKILLS_LIST.find(s => s.learning);
    if (sk && sk.durationSec > 0) sk.durationSec--;
  }

  // Scale & Thriller timers
  if (p.scaleTimer > 0) {
    p.scaleTimer--;
    if (p.scaleTimer === 0) p.scale = 1.0;
  }
  if (p.thrillerTimer > 0) p.thrillerTimer--;
  if (p.musicIconTimer > 0) p.musicIconTimer--;

  // Waiting Room countdown in Andra Bureaucratic Hall (Video B 04:45 - 05:06)
  if (State.waitingRoom.active) {
    if (State.keys.left || State.keys.right || State.keys.jump) {
      addFloatingText(p.x, p.y - 70, "Do not move, just wait!", "#e67e22");
      State.keys.left = false; State.keys.right = false; State.keys.jump = false;
    }
    State.waitingRoom.timer--;
    const secs = Math.ceil(State.waitingRoom.timer / 60);
    document.getElementById("waiting-timer-text").textContent = `at max, ${secs} secs`;
    if (State.waitingRoom.timer <= 0) {
      State.waitingRoom.active = false;
      State.waitingRoom.qualified = true;
      document.getElementById("waiting-banner").style.display = "none";
      const mag = st.npcs.find(n => n.id === "mag_piks");
      if (mag) {
        mag.x = 760;
        State.chatBubbles[mag.id] = { text: `${p.name} Step up`, timer: 240 };
      }
      addChatMessage("Magister Piksnikson", `${p.name}! Step up to receive your Card-Carrying Qualification.`, "reward");
    }
  }

  // Horizontal movement
  const moveSpeed = 4.2;
  if (!State.waitingRoom.active) {
    if (State.keys.left) {
      p.vx = -moveSpeed;
      p.dir = -1;
    } else if (State.keys.right) {
      p.vx = moveSpeed;
      p.dir = 1;
    } else {
      p.vx *= 0.72;
      if (Math.abs(p.vx) < 0.1) p.vx = 0;
    }
  }

  // Ladders check
  const onLadder = st.ladders.find(l => Math.abs(p.x - l.x) < 24 && p.y >= l.y1 && p.y <= l.y2 + 10);
  if (onLadder && (State.keys.up || State.keys.down)) {
    p.climbing = true;
    p.vy = State.keys.up ? -3.5 : 3.5;
  } else {
    p.climbing = false;
  }

  // Jump
  if (State.keys.jump && p.onGround && !State.waitingRoom.active) {
    p.vy = -10.8;
    p.onGround = false;
    State.keys.jump = false;
  }

  // Apply gravity & velocity
  if (!p.climbing) {
    p.vy += 0.48; // gravity
  }
  const prevY = p.y;
  p.x += p.vx;
  p.y += p.vy;

  // Platform collisions (one-way from above)
  p.onGround = false;
  if (p.vy >= 0 && !State.keys.down) {
    for (const plat of st.platforms) {
      if (p.x >= plat.x && p.x <= plat.x + plat.w && prevY <= plat.y + 4 && p.y >= plat.y) {
        p.y = plat.y;
        p.vy = 0;
        p.onGround = true;
        break;
      }
    }
  }

  // Ground collision
  if (p.y >= st.groundY) {
    p.y = st.groundY;
    p.vy = 0;
    p.onGround = true;
  }

  // Street boundaries & automatic signpost exit trigger
  if (p.x < 35) {
    p.x = 35;
    const leftExit = st.exits.find(ex => ex.x < 250);
    if (leftExit) changeStreet(leftExit.target, leftExit.spawnX);
  } else if (p.x > st.width - 35) {
    p.x = st.width - 35;
    const rightExit = st.exits.find(ex => ex.x > st.width - 250);
    if (rightExit) changeStreet(rightExit.target, rightExit.spawnX);
  }

  // Animation state
  if (p.animTimer > 0) {
    p.animTimer--;
  } else if (p.thrillerTimer > 0) {
    p.anim = "thriller";
  } else if (!p.onGround) {
    p.anim = "jump";
  } else if (Math.abs(p.vx) > 0.4) {
    p.anim = "walk";
  } else {
    p.anim = "idle";
  }

  // Quoin collection (Video B 06:16 - 06:26)
  st.quoins.forEach(q => {
    if (q.collected) {
      q.respawnTimer--;
      if (q.respawnTimer <= 0) q.collected = false;
      return;
    }
    if (Math.hypot(p.x - q.x, (p.y - 35) - q.y) < 36) {
      q.collected = true;
      q.respawnTimer = 600;
      AudioEngine.playChime("quoin");
      if (q.kind === "mood") {
        changeMeters({ mood: q.amount });
        addFloatingText(q.x, q.y, `+${q.amount} mood`, "#8de969");
      } else if (q.kind === "xp" || q.kind === "img" || q.kind === "qurazy") {
        changeMeters({ img: q.amount });
        addFloatingText(q.x, q.y, `+${q.amount} xp`, "#b388ff");
      } else if (q.kind === "currants") {
        changeMeters({ currants: q.amount });
        addFloatingText(q.x, q.y, `+${q.amount} currants`, "#f1c40f");
      } else if (q.kind === "energy") {
        changeMeters({ energy: q.amount });
        addFloatingText(q.x, q.y, `+${q.amount} energy`, "#4da6ff");
      }
    }
  });

  // Update Conga Line Followers & Bots
  p.followers.forEach((fId, idx) => {
    const bot = State.otherPlayers[fId];
    if (bot && bot.street === State.streetId) {
      const targetX = p.x - p.dir * (idx + 1) * 44;
      const dx = targetX - bot.x;
      if (Math.abs(dx) > 6) {
        bot.x += Math.sign(dx) * Math.min(Math.abs(dx), 4.0);
        bot.dir = Math.sign(dx) || p.dir;
      }
      bot.y = p.y;
    }
  });

  Object.values(State.otherPlayers).forEach(op => {
    if (op.thrillerTimer > 0) op.thrillerTimer--;
    if (op.isBot && op.street === State.streetId && !p.followers.includes(op.id)) {
      op.actionCooldown--;
      if (op.actionCooldown <= 0) {
        op.actionCooldown = 180 + Math.floor(seededRandom() * 240);
        const step = (seededRandom() - 0.5) * 90;
        op.x = Math.max(140, Math.min(st.width - 140, op.x + step));
        op.dir = step >= 0 ? 1 : -1;
      }
    }
  });

  // Update Cubimals
  for (let i = State.cubimals.length - 1; i >= 0; i--) {
    const c = State.cubimals[i];
    c.x += c.vx;
    c.timer--;
    if (c.timer <= 0) State.cubimals.splice(i, 1);
  }

  // Update Particles & Floating Texts & Chat Bubbles
  for (let i = State.particles.length - 1; i >= 0; i--) {
    const pt = State.particles[i];
    pt.x += pt.vx;
    pt.y += pt.vy;
    pt.vy += 0.06;
    pt.life--;
    if (pt.life <= 0) State.particles.splice(i, 1);
  }
  for (let i = State.floatingTexts.length - 1; i >= 0; i--) {
    const ft = State.floatingTexts[i];
    ft.y -= 0.7;
    ft.life--;
    if (ft.life <= 0) State.floatingTexts.splice(i, 1);
  }
  Object.keys(State.chatBubbles).forEach(k => {
    State.chatBubbles[k].timer--;
    if (State.chatBubbles[k].timer <= 0) delete State.chatBubbles[k];
  });

  // Broadcast player position periodically
  if (simTick % 4 === 0 && (Math.abs(p.vx) > 0.1 || Math.abs(p.vy) > 0.1)) {
    sendNet({
      type: "move",
      id: State.playerId,
      name: p.name,
      street: State.streetId,
      x: Math.round(p.x),
      y: Math.round(p.y),
      dir: p.dir,
      scale: p.scale,
      appearance: p.appearance
    });
  }
}

// --- Procedural 2D Canvas Rendering (Parallax, Props, Avatars, UI) ---
function resizeCanvas() {
  const wrap = document.getElementById("viewport-wrap");
  canvas.width = wrap.clientWidth;
  canvas.height = wrap.clientHeight;
}
window.addEventListener("resize", resizeCanvas);

function renderScene() {
  if (canvas.width === 0) resizeCanvas();
  const W = canvas.width;
  const H = canvas.height;
  const st = STREETS[State.streetId];
  const p = State.player;

  // Smooth camera tracking
  const targetCamX = Math.max(0, Math.min(st.width - W, p.x - W / 2));
  const targetCamY = Math.max(-120, Math.min(80, (p.y - st.groundY) * 0.4 - (H - 540) * 0.5));
  State.camera.x += (targetCamX - State.camera.x) * 0.14;
  State.camera.y += (targetCamY - State.camera.y) * 0.14;

  const camX = State.camera.x;
  const camY = State.camera.y;

  // 1. Sky & Parallax Background Layers by Street Style
  drawParallaxBackground(ctx, W, H, st, camX, camY);

  ctx.save();
  ctx.translate(-camX, -camY);

  // 2. Ground Plane
  drawStreetGround(ctx, st);

  // 3. Street Architecture (Beau's Crooked House / Ministry of Departments / Office Interior)
  drawStreetBuildings(ctx, st);

  // 4. Platforms & Ladders
  st.platforms.forEach(plat => {
    if (st.id !== "beau_home") {
      ctx.fillStyle = st.style === "cavern" ? "#5c4d3c" : "#877c58";
      ctx.beginPath();
      ctx.roundRect(plat.x, plat.y, plat.w, 22, 8);
      ctx.fill();
      ctx.fillStyle = st.style === "cavern" ? "#7a6752" : "#9cb053";
      ctx.fillRect(plat.x + 4, plat.y, plat.w - 8, 6);
    }
  });
  st.ladders.forEach(lad => {
    ctx.strokeStyle = "#8c6239";
    ctx.lineWidth = 4;
    ctx.beginPath();
    ctx.moveTo(lad.x - 10, lad.y1); ctx.lineTo(lad.x - 10, lad.y2);
    ctx.moveTo(lad.x + 10, lad.y1); ctx.lineTo(lad.x + 10, lad.y2);
    for (let ry = lad.y1 + 12; ry < lad.y2; ry += 16) {
      ctx.moveTo(lad.x - 10, ry); ctx.lineTo(lad.x + 10, ry);
    }
    ctx.stroke();
  });

  // 5. Signposts (Exits)
  st.exits.forEach(ex => drawSignpost(ctx, ex));

  // 6. Resources & Projects
  st.resources.forEach(res => drawResource(ctx, res));

  // 7. NPCs & Shrines & Vendors
  st.npcs.forEach(npc => drawNPC(ctx, npc));

  // 8. Floating Quoins
  st.quoins.forEach(q => {
    if (q.collected) return;
    const bob = Math.sin(simTick * 0.08 + q.x) * 5;
    ctx.save();
    ctx.translate(q.x, q.y + bob);
    ctx.strokeStyle = q.kind === "qurazy" ? "#d580ff" : "#ffffff";
    ctx.lineWidth = 2.5;
    ctx.fillStyle = q.kind === "qurazy" ? "rgba(180,80,255,0.45)" : "rgba(255,255,255,0.28)";
    ctx.beginPath();
    ctx.arc(0, 0, q.kind === "qurazy" ? 18 : 14, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();
    ctx.fillStyle = "#fff";
    ctx.font = "bold 11px sans-serif";
    ctx.textAlign = "center";
    ctx.fillText(q.kind === "qurazy" ? "❗" : "✦", 0, 4);
    ctx.restore();
  });

  // 9. Cubimals Racing
  State.cubimals.forEach(c => {
    ctx.save();
    ctx.translate(c.x, c.y - 12);
    ctx.fillStyle = c.kind === "chick_cubimal" ? "#f1c40f" : "#bdc3c7";
    ctx.fillRect(-10, -10, 20, 20);
    ctx.strokeStyle = "#333";
    ctx.strokeRect(-10, -10, 20, 20);
    ctx.restore();
  });

  // 10. Other Players / Bots on this Street
  Object.values(State.otherPlayers).forEach(op => {
    if (op.street === State.streetId) {
      drawGlitchenAvatar(ctx, op, false);
    }
  });

  // 11. Local Player Avatar
  drawGlitchenAvatar(ctx, {
    id: State.playerId,
    name: p.name,
    x: p.x,
    y: p.y,
    dir: p.dir,
    anim: p.anim,
    scale: p.scale,
    thrillerTimer: p.thrillerTimer,
    musicIconTimer: p.musicIconTimer,
    appearance: p.appearance
  }, true);

  // 12. Particles
  State.particles.forEach(pt => {
    ctx.save();
    ctx.fillStyle = pt.color;
    ctx.translate(pt.x, pt.y);
    ctx.beginPath();
    ctx.arc(0, 0, pt.kind === "sparkle" ? 4 : 5, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
  });

  // 13. Floating Texts
  State.floatingTexts.forEach(ft => {
    ctx.save();
    ctx.font = "bold 14px 'Trebuchet MS', sans-serif";
    ctx.textAlign = "center";
    ctx.fillStyle = "rgba(0,0,0,0.65)";
    ctx.fillText(ft.text, ft.x + 1, ft.y + 1);
    ctx.fillStyle = ft.color;
    ctx.fillText(ft.text, ft.x, ft.y);
    ctx.restore();
  });

  ctx.restore();

  // 14. Update Minimap
  drawMinimap(st);
}

function drawParallaxBackground(ctx, W, H, st, camX, camY) {
  if (st.style === "lost_woods") {
    // Spooky dark forest from Video A 00:00
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, "#3b4642");
    grad.addColorStop(0.65, "#111615");
    grad.addColorStop(1, "#050707");
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // Distant silhouetted bare trees (parallax 0.25)
    ctx.fillStyle = "#18201e";
    for (let i = 0; i < 12; i++) {
      const tx = ((i * 260 - camX * 0.25) % (W + 400)) - 100;
      ctx.fillRect(tx, 60, 26, H);
    }
    // Midground gnarled dark trees (parallax 0.5)
    ctx.fillStyle = "#0b0e0d";
    for (let i = 0; i < 9; i++) {
      const tx = ((i * 340 - camX * 0.5) % (W + 500)) - 120;
      ctx.fillRect(tx, 40, 42, H);
      ctx.beginPath();
      ctx.moveTo(tx + 20, 180);
      ctx.lineTo(tx - 60, 90);
      ctx.lineTo(tx - 45, 80);
      ctx.lineTo(tx + 25, 150);
      ctx.fill();
    }
  } else if (st.style === "vorn_road") {

function drawGlitchenAvatar(ctx, av, isSelf) {
  const ap = av.appearance || {
    skinColor: av.skinColor || "#fce2c4",
    hairColor: av.hairColor || "#f5e06b",
    shirtColor: av.shirtColor || "#1f2428",
    pantsColor: av.pantsColor || "#1a1d20",
    shoeColor: "#e74c3c",
    hat: av.hat || "none",
    eyeScale: 1.0,
    heightScale: 1.0
  };
  const scale = (av.scale || 1.0) * (ap.heightScale || 1.0);
  const isThriller = (av.thrillerTimer || 0) > 0;
  const isWalking = av.anim === "walk";
  const walkCycle = isWalking ? Math.sin(simTick * 0.28) : 0;
  const thrillerCycle = isThriller ? Math.sin(simTick * 0.25) : 0;

  ctx.save();
  ctx.translate(av.x, av.y);

  // Blue aura glow during Thriller or buff (Video A 29:57)

    ctx.beginPath();
    ctx.ellipse(0, -38 * scale, 28 * scale, 46 * scale, 0, 0, Math.PI * 2);

  ctx.scale((av.dir || 1) * scale, scale);

  ctx.strokeStyle = ap.pantsColor;

  ctx.moveTo(-5, -18); ctx.lineTo(-5 + walkCycle * 7, 0);

/**
 * Giants' Reverie — Zero-Dependency Node 22 Server (HTTP + RFC 6455 WebSocket)
 * Uses only built-in Node modules: http, crypto, fs, path.
 */
const http = require("http");
const crypto = require("crypto");
const fs = require("fs");
const path = require("path");

const PORT = parseInt(process.env.PORT || "8090", 10);
const PUBLIC_DIR = path.join(__dirname, "public");

const MIME_TYPES = {
  ".html": "text/html; charset=utf-8",
  ".js": "application/javascript; charset=utf-8",
  ".css": "text/css; charset=utf-8",
  ".json": "application/json; charset=utf-8",
  ".png": "image/png",
  ".svg": "image/svg+xml"
};

// --- Shared Multiplayer State ---
const clients = new Map(); // socket -> player state
const recentMines = new Map(); // `${street}:${rockId}` -> { playerId, time }
let towerProgress = 11.4;
const streetStyles = {
  beau_home: "lost_woods"

// --- HTTP Static File Server ---
const server = http.createServer((req, res) => {
  const safeUrl = (req.url || "/").split("?")[0];
  const relPath = safeUrl === "/" ? "/index.html" : safeUrl;
  const filePath = path.normalize(path.join(PUBLIC_DIR, relPath));

  if (!filePath.startsWith(PUBLIC_DIR)) {
    res.writeHead(403);
    res.end("Forbidden");
    return;
  }

  fs.readFile(filePath, (err, content) => {
    if (err) {
      res.writeHead(404, { "Content-Type": "text/plain; charset=utf-8" });
      res.end("Not found");
      return;
    }
    const ext = path.extname(filePath).toLowerCase();
    res.writeHead(200, { "Content-Type": MIME_TYPES[ext] || "application/octet-stream" });
    res.end(content);
  });

// --- Raw RFC 6455 WebSocket Implementation ---
const WS_GUID = "258EAFA5-E914-47DA-95CA-C5AB0DC85B11";

function encodeWsFrame(str) {
  const payload = Buffer.from(str, "utf8");
  const len = payload.length;
  let header;
  if (len < 126) {
    header = Buffer.alloc(2);
    header[0] = 0x81; // FIN + text frame
    header[1] = len;
  } else if (len < 65536) {
    header = Buffer.alloc(4);
    header[0] = 0x81;
    header[1] = 126;
    header.writeUInt16BE(len, 2);
  } else {
    header = Buffer.alloc(10);
    header[0] = 0x81;
    header[1] = 127;
    header.writeBigUInt64BE(BigInt(len), 2);
  }
  return Buffer.concat([header, payload]);
}

function decodeWsFrames(buffer) {
  const messages = [];
  let offset = 0;

  while (offset + 2 <= buffer.length) {
    const byte0 = buffer[offset];
    const byte1 = buffer[offset + 1];
    const opcode = byte0 & 0x0f;
    const isMasked = (byte1 & 0x80) !== 0;
    let payloadLen = byte1 & 0x7f;
    let headerLen = 2;

    if (opcode === 0x08) {
      return { messages, rest: Buffer.alloc(0), closed: true };
    }

    if (payloadLen === 126) {
      if (offset + 4 > buffer.length) break;
      payloadLen = buffer.readUInt16BE(offset + 2);
      headerLen = 4;
    } else if (payloadLen === 127) {
      if (offset + 10 > buffer.length) break;
      payloadLen = Number(buffer.readBigUInt64BE(offset + 2));
      headerLen = 10;
    }

    const maskLen = isMasked ? 4 : 0;
    const totalFrameLen = headerLen + maskLen + payloadLen;
    if (offset + totalFrameLen > buffer.length) break;

    const payload = Buffer.alloc(payloadLen);
    if (isMasked) {
      const maskOffset = offset + headerLen;
      const dataOffset = maskOffset + 4;
      for (let i = 0; i < payloadLen; i++) {
        payload[i] = buffer[dataOffset + i] ^ buffer[maskOffset + (i & 3)];
      }
    } else {
      buffer.copy(payload, 0, offset + headerLen, offset + headerLen + payloadLen);
    }

    if (opcode === 0x01) {
      messages.push(payload.toString("utf8"));
    }
    offset += totalFrameLen;
  }

  return { messages, rest: buffer.subarray(offset), closed: false };
}

function sendJson(socket, obj) {
  if (socket.destroyed || !socket.writable) return;
  try {
    socket.write(encodeWsFrame(JSON.stringify(obj)));
  } catch (e) {}
}

function broadcastAll(obj, excludeSocket = null) {
  const frame = encodeWsFrame(JSON.stringify(obj));
  for (const [sock] of clients.entries()) {
    if (sock !== excludeSocket && !sock.destroyed && sock.writable) {
      try { sock.write(frame); } catch (e) {}
    }
  }
}

function broadcastStreet(streetId, obj, excludeSocket = null) {
  const frame = encodeWsFrame(JSON.stringify(obj));
  for (const [sock, p] of clients.entries()) {
    if (sock !== excludeSocket && p.street === streetId && !sock.destroyed && sock.writable) {
      try { sock.write(frame); } catch (e) {}
    }
  }
}

function getPlayersOnStreet(streetId) {
  const arr = [];
  for (const p of clients.values()) {
    if (p.street === streetId) arr.push(p);
  }
  return arr;
}

server.on("upgrade", (req, socket) => {
  const key = req.headers["sec-websocket-key"];
  if (!key) {
    socket.destroy();
    return;
  }
  const accept = crypto.createHash("sha1").update(key + WS_GUID).digest("base64");
  const headers = [
    "HTTP/1.1 101 Switching Protocols",
    "Upgrade: websocket",
    "Connection: Upgrade",
    `Sec-WebSocket-Accept: ${accept}`,
    "\r\n"
  ];
  socket.write(headers.join("\r\n"));

  let rxBuffer = Buffer.alloc(0);
  clients.set(socket, {
    id: "anon_" + Math.random().toString(36).slice(2, 8),
    name: "Glitchen",
    street: "beau_home",
    x: 620,
    y: 460,
    dir: 1,
    scale: 1.0,
    appearance: {}
  });

  socket.on("data", (chunk) => {
    rxBuffer = Buffer.concat([rxBuffer, chunk]);
    const decoded = decodeWsFrames(rxBuffer);
    rxBuffer = decoded.rest;

    for (const raw of decoded.messages) {
      try {
        const msg = JSON.parse(raw);
        handleClientMessage(socket, msg);
      } catch (e) {}
    }

    if (decoded.closed) {
      cleanupSocket(socket);
      socket.end();
    }
  });

  socket.on("close", () => cleanupSocket(socket));
  socket.on("error", () => cleanupSocket(socket));
});

function handleClientMessage(socket, msg) {
  const p = clients.get(socket);
  if (!p) return;

  if (msg.type === "join") {
    p.id = msg.id || p.id;
    p.name = msg.name || p.name;
    p.street = msg.street || "beau_home";
    p.x = msg.x || 620;
    p.y = msg.y || 460;
    p.dir = msg.dir || 1;
    p.scale = msg.scale || 1.0;
    p.appearance = msg.appearance || {};

    sendJson(socket, { type: "street_players", players: getPlayersOnStreet(p.street) });
    sendJson(socket, { type: "tower_progress", progress: towerProgress });
    if (streetStyles[p.street]) {
      sendJson(socket, { type: "street_style", street: p.street, style: streetStyles[p.street] });
    }
    broadcastStreet(p.street, { type: "street_players", players: getPlayersOnStreet(p.street) }, socket);
    broadcastStreet(p.street, { type: "chat", sender: null, text: `${p.name} is here!`, kind: "sys" }, socket);
  } else if (msg.type === "move") {
    p.x = msg.x;
    p.y = msg.y;
    p.dir = msg.dir;
    p.scale = msg.scale || 1.0;
    if (msg.appearance) p.appearance = msg.appearance;
    broadcastStreet(p.street, {
      type: "player_moved",
      id: p.id,
      name: p.name,
      street: p.street,
      x: p.x,
      y: p.y,
      dir: p.dir,
      scale: p.scale,
      appearance: p.appearance
    }, socket);
  } else if (msg.type === "change_street") {
    const oldStreet = p.street;
    p.street = msg.street;
    p.x = msg.x || 300;
    p.y = msg.y || 460;
    broadcastStreet(oldStreet, { type: "player_left", id: p.id }, socket);
    sendJson(socket, { type: "street_players", players: getPlayersOnStreet(p.street) });
    broadcastStreet(p.street, { type: "street_players", players: getPlayersOnStreet(p.street) }, socket);
  } else if (msg.type === "chat") {
    broadcastStreet(p.street, {
      type: "chat",
      sender: p.name,
      senderId: p.id,
      text: String(msg.text || "").slice(0, 140),
      kind: "normal"
    }, socket);
  } else if (msg.type === "emote") {
    broadcastStreet(p.street, {
      type: "emote",
      kind: msg.kind,
      sender: p.name,
      x: p.x,
      y: p.y
    }, socket);
  } else if (msg.type === "mine") {
    const key = `${p.street}:${msg.rockId}`;
    const prev = recentMines.get(key);
    const now = Date.now();
    if (prev && prev.playerId !== p.id && now - prev.time < 6000) {
      sendJson(socket, { type: "co_mine_bonus" });
    }
    recentMines.set(key, { playerId: p.id, time: now });
  } else if (msg.type === "tower_progress") {
    towerProgress = Math.min(100, Number(msg.progress) || towerProgress);
    broadcastAll({ type: "tower_progress", progress: towerProgress });
  } else if (msg.type === "street_style") {
    streetStyles[msg.street] = msg.style;
    broadcastAll({ type: "street_style", street: msg.street, style: msg.style }, socket);
  } else if (msg.type === "giant_visage") {
    broadcastAll({ type: "giant_visage", giant: msg.giant }, socket);
  }
}

function cleanupSocket(socket) {
  const p = clients.get(socket);
  if (p) {
    clients.delete(socket);
    broadcastStreet(p.street, { type: "player_left", id: p.id });
  }
}

# Giants' Reverie

A 2D side-scrolling multiplayer browser MMO rebuilt from the surviving gameplay footage of *Glitch* (Videos A & B).

## How to Run

1. Ensure **Node.js 22+** is installed (zero external dependencies required).
2. Start the server:
   ```bash
   node server.js
   ```
   *(Set `PORT=3000` environment variable to override the default port `8090`.)*
3. Open `http://localhost:8090` in one or more browser windows (pass `?name=DeadSparrow&seed=42` in the URL to customize your name and deterministic RNG seed).
4. **Solo Mode:** If `public/index.html` is opened directly or the server is unreachable, the client runs solo with simulated Glitchen companions on the streets.

## Controls

- **Move Left / Right:** `A` / `D` or `Left` / `Right` Arrow keys
- **Jump:** `Space`, `W`, or `Up` Arrow key (jump onto house roof ledges, rock platforms, and floating Quoins)
- **Climb Ladders / Drop Through Platforms:** `W` / `S` or `Up` / `Down` Arrow keys
- **Interact with Trees, Rocks, Shrines, Vendors, Bureaucrats, or Players:** Click directly on any object or avatar in the world to open its verb menu
- **Use Inventory Items:** Click any item in the 16-slot bottom inventory bar to open its context verbs (`Drink`, `Eat`, `Taste`, `Bestow`, `Sprinkle`, `Play`, `Roll`, `Set out`, `Open/Close Bag`, `Drop`, `Inspect`)
- **Chat & Commands:** Type in the bottom-right chat box and press `Enter` (try `/thriller` or `/poof`)
